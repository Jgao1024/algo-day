# 最长公共子串

## 题目描述
给定两个字符串，求它们的最长公共子串的长度。

子串是连续的字符序列，而子序列可以不连续。

示例：
```
输入：str1 = "ABCDGH", str2 = "ACDGHR"
输出：4
解释：最长公共子串是 "CDGH"，长度为 4。

输入：str1 = "ABC", str2 = "DEF"
输出：0
解释：没有公共子串，返回 0。
```

## 解题思路
1. 动态规划方法：
   - 定义dp[i][j]表示以str1[i-1]和str2[j-1]结尾的最长公共子串长度
   - 状态转移方程：
     - 如果str1[i-1] == str2[j-1]，dp[i][j] = dp[i-1][j-1] + 1
     - 否则，dp[i][j] = 0
   - 时间复杂度：O(mn)，空间复杂度：O(mn)

2. 空间优化方法：
   - 使用滚动数组优化空间
   - 时间复杂度：O(mn)，空间复杂度：O(min(m,n))

3. 后缀数组方法：
   - 将两个字符串连接起来
   - 构建后缀数组
   - 时间复杂度：O(n log n)，空间复杂度：O(n)

## Java实现
```java
class Solution {
    // 动态规划方法
    public int longestCommonSubstring(String str1, String str2) {
        int m = str1.length();
        int n = str2.length();
        int[][] dp = new int[m + 1][n + 1];
        int maxLen = 0;
        
        for (int i = 1; i <= m; i++) {
            for (int j = 1; j <= n; j++) {
                if (str1.charAt(i - 1) == str2.charAt(j - 1)) {
                    dp[i][j] = dp[i - 1][j - 1] + 1;
                    maxLen = Math.max(maxLen, dp[i][j]);
                }
            }
        }
        
        return maxLen;
    }
    
    // 空间优化方法
    public int longestCommonSubstringOptimized(String str1, String str2) {
        int m = str1.length();
        int n = str2.length();
        
        // 确保str1是较短的字符串
        if (m > n) {
            return longestCommonSubstringOptimized(str2, str1);
        }
        
        int[] dp = new int[m + 1];
        int maxLen = 0;
        
        for (int j = 1; j <= n; j++) {
            int prev = 0;
            for (int i = 1; i <= m; i++) {
                int temp = dp[i];
                if (str1.charAt(i - 1) == str2.charAt(j - 1)) {
                    dp[i] = prev + 1;
                    maxLen = Math.max(maxLen, dp[i]);
                } else {
                    dp[i] = 0;
                }
                prev = temp;
            }
        }
        
        return maxLen;
    }
    
    // 后缀数组方法
    public int longestCommonSubstringSuffixArray(String str1, String str2) {
        String combined = str1 + "#" + str2;
        int n = combined.length();
        int[] sa = buildSuffixArray(combined);
        int[] lcp = buildLCP(combined, sa);
        
        int maxLen = 0;
        for (int i = 1; i < n; i++) {
            if ((sa[i] < str1.length() && sa[i-1] > str1.length()) ||
                (sa[i] > str1.length() && sa[i-1] < str1.length())) {
                maxLen = Math.max(maxLen, lcp[i]);
            }
        }
        
        return maxLen;
    }
    
    private int[] buildSuffixArray(String s) {
        int n = s.length();
        int[] sa = new int[n];
        int[] rank = new int[n];
        
        // 初始化
        for (int i = 0; i < n; i++) {
            sa[i] = i;
            rank[i] = s.charAt(i);
        }
        
        // 基数排序
        for (int k = 1; k < n; k *= 2) {
            // 对第二关键字排序
            int[] temp = new int[n];
            for (int i = 0; i < n; i++) {
                temp[i] = (i + k < n) ? rank[i + k] : -1;
            }
            
            // 对第一关键字排序
            int[] count = new int[Math.max(n, 256)];
            for (int i = 0; i < n; i++) {
                count[temp[i] + 1]++;
            }
            for (int i = 1; i < count.length; i++) {
                count[i] += count[i - 1];
            }
            for (int i = n - 1; i >= 0; i--) {
                sa[count[temp[i] + 1] - 1] = i;
                count[temp[i] + 1]--;
            }
            
            // 更新rank
            int[] newRank = new int[n];
            newRank[sa[0]] = 0;
            for (int i = 1; i < n; i++) {
                newRank[sa[i]] = newRank[sa[i - 1]];
                if (rank[sa[i]] != rank[sa[i - 1]] ||
                    (sa[i] + k < n ? rank[sa[i] + k] : -1) !=
                    (sa[i - 1] + k < n ? rank[sa[i - 1] + k] : -1)) {
                    newRank[sa[i]]++;
                }
            }
            rank = newRank;
        }
        
        return sa;
    }
    
    private int[] buildLCP(String s, int[] sa) {
        int n = s.length();
        int[] rank = new int[n];
        for (int i = 0; i < n; i++) {
            rank[sa[i]] = i;
        }
        
        int[] lcp = new int[n];
        int k = 0;
        for (int i = 0; i < n; i++) {
            if (rank[i] == n - 1) {
                k = 0;
                continue;
            }
            
            int j = sa[rank[i] + 1];
            while (i + k < n && j + k < n && s.charAt(i + k) == s.charAt(j + k)) {
                k++;
            }
            lcp[rank[i]] = k;
            if (k > 0) {
                k--;
            }
        }
        
        return lcp;
    }
}
```

## Python实现
```python
class Solution:
    # 动态规划方法
    def longestCommonSubstring(self, str1: str, str2: str) -> int:
        m, n = len(str1), len(str2)
        dp = [[0] * (n + 1) for _ in range(m + 1)]
        max_len = 0
        
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if str1[i - 1] == str2[j - 1]:
                    dp[i][j] = dp[i - 1][j - 1] + 1
                    max_len = max(max_len, dp[i][j])
        
        return max_len
    
    # 空间优化方法
    def longestCommonSubstringOptimized(self, str1: str, str2: str) -> int:
        m, n = len(str1), len(str2)
        
        # 确保str1是较短的字符串
        if m > n:
            return self.longestCommonSubstringOptimized(str2, str1)
        
        dp = [0] * (m + 1)
        max_len = 0
        
        for j in range(1, n + 1):
            prev = 0
            for i in range(1, m + 1):
                temp = dp[i]
                if str1[i - 1] == str2[j - 1]:
                    dp[i] = prev + 1
                    max_len = max(max_len, dp[i])
                else:
                    dp[i] = 0
                prev = temp
        
        return max_len
    
    # 后缀数组方法
    def longestCommonSubstringSuffixArray(self, str1: str, str2: str) -> int:
        combined = str1 + "#" + str2
        n = len(combined)
        sa = self.build_suffix_array(combined)
        lcp = self.build_lcp(combined, sa)
        
        max_len = 0
        for i in range(1, n):
            if (sa[i] < len(str1) and sa[i-1] > len(str1)) or \
               (sa[i] > len(str1) and sa[i-1] < len(str1)):
                max_len = max(max_len, lcp[i])
        
        return max_len
    
    def build_suffix_array(self, s: str) -> List[int]:
        n = len(s)
        sa = list(range(n))
        rank = [ord(c) for c in s]
        
        k = 1
        while k < n:
            # 对第二关键字排序
            temp = [rank[i + k] if i + k < n else -1 for i in range(n)]
            
            # 对第一关键字排序
            count = [0] * (max(n, 256) + 1)
            for x in temp:
                count[x + 1] += 1
            for i in range(1, len(count)):
                count[i] += count[i - 1]
            
            new_sa = [0] * n
            for i in range(n - 1, -1, -1):
                new_sa[count[temp[i] + 1] - 1] = sa[i]
                count[temp[i] + 1] -= 1
            sa = new_sa
            
            # 更新rank
            new_rank = [0] * n
            new_rank[sa[0]] = 0
            for i in range(1, n):
                new_rank[sa[i]] = new_rank[sa[i - 1]]
                if rank[sa[i]] != rank[sa[i - 1]] or \
                   (sa[i] + k < n and sa[i - 1] + k < n and
                    rank[sa[i] + k] != rank[sa[i - 1] + k]):
                    new_rank[sa[i]] += 1
            rank = new_rank
            
            k *= 2
        
        return sa
    
    def build_lcp(self, s: str, sa: List[int]) -> List[int]:
        n = len(s)
        rank = [0] * n
        for i in range(n):
            rank[sa[i]] = i
        
        lcp = [0] * n
        k = 0
        for i in range(n):
            if rank[i] == n - 1:
                k = 0
                continue
            
            j = sa[rank[i] + 1]
            while i + k < n and j + k < n and s[i + k] == s[j + k]:
                k += 1
            lcp[rank[i]] = k
            if k > 0:
                k -= 1
        
        return lcp
```

## 复杂度分析
- 动态规划方法：
  - 时间复杂度：O(mn)，其中m和n分别是两个字符串的长度
  - 空间复杂度：O(mn)
- 空间优化方法：
  - 时间复杂度：O(mn)
  - 空间复杂度：O(min(m,n))
- 后缀数组方法：
  - 时间复杂度：O(n log n)
  - 空间复杂度：O(n)
