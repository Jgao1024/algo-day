# 76. 最小覆盖子串

## 题目描述
给你一个字符串 s 和一个字符串 t 。返回 s 中涵盖 t 所有字符的最小子串。如果 s 中不存在涵盖 t 所有字符的子串，则返回空字符串 "" 。

注意：
- 对于 t 中重复字符，我们寻找的子字符串中该字符数量必须不少于 t 中该字符数量。
- 如果 s 中存在这样的子串，我们保证它是唯一的答案。

示例：
```
输入：s = "ADOBECODEBANC", t = "ABC"
输出："BANC"
解释：最小覆盖子串 "BANC" 包含来自字符串 t 的 'A'、'B' 和 'C'。

输入：s = "a", t = "a"
输出："a"
解释：整个字符串 s 是最小覆盖子串。

输入：s = "a", t = "aa"
输出：""
解释：t 中两个字符 'a' 均应包含在 s 的子串中，因此没有符合条件的子字符串，返回空字符串。
```

## 解题思路
1. 滑动窗口方法：
   - 使用可变大小的滑动窗口
   - 使用哈希表记录字符出现次数
   - 维护窗口内字符的匹配状态
   - 时间复杂度：O(n)，空间复杂度：O(k)

2. 关键点：
   - 窗口大小可变
   - 字符频率统计
   - 匹配状态维护
   - 最小子串更新

## Java实现
```java
class Solution {
    public String minWindow(String s, String t) {
        if (s.length() < t.length()) {
            return "";
        }
        
        // 统计t中字符出现次数
        int[] count = new int[128];
        for (char c : t.toCharArray()) {
            count[c]++;
        }
        
        int left = 0, right = 0;
        int minLen = Integer.MAX_VALUE;
        int minStart = 0;
        int matches = 0;
        
        while (right < s.length()) {
            char c = s.charAt(right);
            count[c]--;
            
            if (count[c] >= 0) {
                matches++;
            }
            
            // 当窗口包含所有t中的字符时，尝试收缩窗口
            while (matches == t.length()) {
                if (right - left + 1 < minLen) {
                    minLen = right - left + 1;
                    minStart = left;
                }
                
                char leftChar = s.charAt(left);
                count[leftChar]++;
                
                if (count[leftChar] > 0) {
                    matches--;
                }
                
                left++;
            }
            
            right++;
        }
        
        return minLen == Integer.MAX_VALUE ? "" : s.substring(minStart, minStart + minLen);
    }
    
    // 使用HashMap的实现方式
    public String minWindowHashMap(String s, String t) {
        if (s.length() < t.length()) {
            return "";
        }
        
        Map<Character, Integer> need = new HashMap<>();
        Map<Character, Integer> window = new HashMap<>();
        
        // 统计t中字符出现次数
        for (char c : t.toCharArray()) {
            need.put(c, need.getOrDefault(c, 0) + 1);
        }
        
        int left = 0, right = 0;
        int minLen = Integer.MAX_VALUE;
        int minStart = 0;
        int matches = 0;
        
        while (right < s.length()) {
            char c = s.charAt(right);
            window.put(c, window.getOrDefault(c, 0) + 1);
            
            if (need.containsKey(c) && window.get(c).equals(need.get(c))) {
                matches++;
            }
            
            while (matches == need.size()) {
                if (right - left + 1 < minLen) {
                    minLen = right - left + 1;
                    minStart = left;
                }
                
                char leftChar = s.charAt(left);
                window.put(leftChar, window.get(leftChar) - 1);
                
                if (need.containsKey(leftChar) && window.get(leftChar) < need.get(leftChar)) {
                    matches--;
                }
                
                left++;
            }
            
            right++;
        }
        
        return minLen == Integer.MAX_VALUE ? "" : s.substring(minStart, minStart + minLen);
    }
}
```

## Python实现
```python
class Solution:
    def minWindow(self, s: str, t: str) -> str:
        if len(s) < len(t):
            return ""
        
        # 统计t中字符出现次数
        count = [0] * 128
        for c in t:
            count[ord(c)] += 1
        
        left = right = 0
        min_len = float('inf')
        min_start = 0
        matches = 0
        
        while right < len(s):
            c = s[right]
            count[ord(c)] -= 1
            
            if count[ord(c)] >= 0:
                matches += 1
            
            # 当窗口包含所有t中的字符时，尝试收缩窗口
            while matches == len(t):
                if right - left + 1 < min_len:
                    min_len = right - left + 1
                    min_start = left
                
                left_char = s[left]
                count[ord(left_char)] += 1
                
                if count[ord(left_char)] > 0:
                    matches -= 1
                
                left += 1
            
            right += 1
        
        return "" if min_len == float('inf') else s[min_start:min_start + min_len]
    
    # 使用字典的实现方式
    def minWindowDict(self, s: str, t: str) -> str:
        if len(s) < len(t):
            return ""
        
        need = {}
        window = {}
        
        # 统计t中字符出现次数
        for c in t:
            need[c] = need.get(c, 0) + 1
        
        left = right = 0
        min_len = float('inf')
        min_start = 0
        matches = 0
        
        while right < len(s):
            c = s[right]
            window[c] = window.get(c, 0) + 1
            
            if c in need and window[c] == need[c]:
                matches += 1
            
            while matches == len(need):
                if right - left + 1 < min_len:
                    min_len = right - left + 1
                    min_start = left
                
                left_char = s[left]
                window[left_char] -= 1
                
                if left_char in need and window[left_char] < need[left_char]:
                    matches -= 1
                
                left += 1
            
            right += 1
        
        return "" if min_len == float('inf') else s[min_start:min_start + min_len]
```

## 复杂度分析
- 基本方法：
  - 时间复杂度：O(n)，其中n是s的长度
  - 空间复杂度：O(1)，因为字符集大小固定
- HashMap方法：
  - 时间复杂度：O(n)
  - 空间复杂度：O(k)，其中k是t中不同字符的数量
