# 460. LFU缓存

## 题目描述
设计并实现最不经常使用（LFU）缓存的数据结构。它应该支持以下操作：get 和 put。

## 解题思路
1. 双哈希表+双链表：
   - 用一个哈希表存储key到节点的映射
   - 用另一个哈希表存储频率到双向链表的映射
   - 每次访问或插入时更新频率
   - 淘汰时移除最小频率链表的头节点
   - 时间复杂度：O(1)

2. 关键点：
   - 频率统计
   - 双向链表维护顺序
   - 哈希表快速定位

## Java实现
```java
class LFUCache {
    class Node {
        int key, value, freq;
        Node prev, next;
        Node(int k, int v) { key = k; value = v; freq = 1; }
    }
    class DLinkedList {
        Node head, tail;
        int size;
        DLinkedList() {
            head = new Node(0, 0);
            tail = new Node(0, 0);
            head.next = tail;
            tail.prev = head;
            size = 0;
        }
        void add(Node node) {
            node.next = head.next;
            node.prev = head;
            head.next.prev = node;
            head.next = node;
            size++;
        }
        void remove(Node node) {
            node.prev.next = node.next;
            node.next.prev = node.prev;
            size--;
        }
        Node removeLast() {
            if (size > 0) {
                Node last = tail.prev;
                remove(last);
                return last;
            }
            return null;
        }
    }
    int capacity, minFreq;
    Map<Integer, Node> keyNode;
    Map<Integer, DLinkedList> freqList;
    public LFUCache(int capacity) {
        this.capacity = capacity;
        this.minFreq = 0;
        keyNode = new HashMap<>();
        freqList = new HashMap<>();
    }
    public int get(int key) {
        if (!keyNode.containsKey(key)) return -1;
        Node node = keyNode.get(key);
        update(node);
        return node.value;
    }
    public void put(int key, int value) {
        if (capacity == 0) return;
        if (keyNode.containsKey(key)) {
            Node node = keyNode.get(key);
            node.value = value;
            update(node);
        } else {
            if (keyNode.size() == capacity) {
                Node toRemove = freqList.get(minFreq).removeLast();
                keyNode.remove(toRemove.key);
            }
            Node node = new Node(key, value);
            keyNode.put(key, node);
            freqList.computeIfAbsent(1, k -> new DLinkedList()).add(node);
            minFreq = 1;
        }
    }
    private void update(Node node) {
        int freq = node.freq;
        freqList.get(freq).remove(node);
        if (freqList.get(freq).size == 0 && freq == minFreq) minFreq++;
        node.freq++;
        freqList.computeIfAbsent(node.freq, k -> new DLinkedList()).add(node);
    }
}
```

## Python实现
```python
class Node:
    def __init__(self, key, value):
        self.key = key
        self.value = value
        self.freq = 1
        self.prev = self.next = None

class DLinkedList:
    def __init__(self):
        self.head = Node(0, 0)
        self.tail = Node(0, 0)
        self.head.next = self.tail
        self.tail.prev = self.head
        self.size = 0
    def add(self, node):
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
        self.size += 1
    def remove(self, node):
        node.prev.next = node.next
        node.next.prev = node.prev
        self.size -= 1
    def remove_last(self):
        if self.size > 0:
            last = self.tail.prev
            self.remove(last)
            return last
        return None

class LFUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.min_freq = 0
        self.key_node = {}
        self.freq_list = {}
    def get(self, key: int) -> int:
        if key not in self.key_node:
            return -1
        node = self.key_node[key]
        self._update(node)
        return node.value
    def put(self, key: int, value: int) -> None:
        if self.capacity == 0:
            return
        if key in self.key_node:
            node = self.key_node[key]
            node.value = value
            self._update(node)
        else:
            if len(self.key_node) == self.capacity:
                to_remove = self.freq_list[self.min_freq].remove_last()
                del self.key_node[to_remove.key]
            node = Node(key, value)
            self.key_node[key] = node
            self.freq_list.setdefault(1, DLinkedList()).add(node)
            self.min_freq = 1
    def _update(self, node):
        freq = node.freq
        self.freq_list[freq].remove(node)
        if self.freq_list[freq].size == 0 and freq == self.min_freq:
            self.min_freq += 1
        node.freq += 1
        self.freq_list.setdefault(node.freq, DLinkedList()).add(node)
```

## 复杂度分析
- 时间复杂度：O(1)（所有操作均摊）
- 空间复杂度：O(capacity)
