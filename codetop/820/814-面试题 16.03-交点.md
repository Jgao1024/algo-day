## 最佳解法思路
- 用向量求两直线交点：若不平行，算出交点后判断是否落在两线段的包围盒内。
- 若平行：  
  - 不共线则无交点  
  - 共线则可能重叠，取重叠部分里最小的点（x 最小，其次 y 最小）。

## Java 最佳实现

```java
import java.util.ArrayList;
import java.util.List;

class Solution {
    private static final double EPS = 1e-7;

    public double[] intersection(int[] start1, int[] end1, int[] start2, int[] end2) {
        double x1 = start1[0], y1 = start1[1];
        double x2 = end1[0], y2 = end1[1];
        double x3 = start2[0], y3 = start2[1];
        double x4 = end2[0], y4 = end2[1];

        double rx = x2 - x1, ry = y2 - y1;
        double sx = x4 - x3, sy = y4 - y3;
        double crossRS = rx * sy - ry * sx;
        double qpx = x3 - x1, qpy = y3 - y1;

        if (Math.abs(crossRS) > EPS) {
            double t = (qpx * sy - qpy * sx) / crossRS;
            double ix = x1 + t * rx;
            double iy = y1 + t * ry;
            if (onSegment(ix, iy, x1, y1, x2, y2) && onSegment(ix, iy, x3, y3, x4, y4)) {
                return new double[]{ix, iy};
            }
            return new double[0];
        }

        if (Math.abs(qpx * ry - qpy * rx) > EPS) {
            return new double[0];
        }

        List<double[]> cand = new ArrayList<>();
        addIfOnBoth(cand, x1, y1, x1, y1, x2, y2, x3, y3, x4, y4);
        addIfOnBoth(cand, x2, y2, x1, y1, x2, y2, x3, y3, x4, y4);
        addIfOnBoth(cand, x3, y3, x1, y1, x2, y2, x3, y3, x4, y4);
        addIfOnBoth(cand, x4, y4, x1, y1, x2, y2, x3, y3, x4, y4);

        if (cand.isEmpty()) return new double[0];

        double[] best = cand.get(0);
        for (int i = 1; i < cand.size(); i++) {
            double[] p = cand.get(i);
            if (less(p, best)) best = p;
        }
        return new double[]{best[0], best[1]};
    }

    private void addIfOnBoth(List<double[]> cand, double px, double py,
                             double x1, double y1, double x2, double y2,
                             double x3, double y3, double x4, double y4) {
        if (onSegment(px, py, x1, y1, x2, y2) && onSegment(px, py, x3, y3, x4, y4)) {
            cand.add(new double[]{px, py});
        }
    }

    private boolean onSegment(double px, double py, double x1, double y1, double x2, double y2) {
        if (Math.abs(cross(x1, y1, x2, y2, px, py)) > EPS) return false;
        return inRange(px, x1, x2) && inRange(py, y1, y2);
    }

    private double cross(double x1, double y1, double x2, double y2, double x3, double y3) {
        return (x2 - x1) * (y3 - y1) - (y2 - y1) * (x3 - x1);
    }

    private boolean inRange(double v, double a, double b) {
        double lo = Math.min(a, b) - EPS;
        double hi = Math.max(a, b) + EPS;
        return v >= lo && v <= hi;
    }

    private boolean less(double[] a, double[] b) {
        if (Math.abs(a[0] - b[0]) > EPS) return a[0] < b[0];
        return a[1] < b[1] - EPS;
    }
}
```
