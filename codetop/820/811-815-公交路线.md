## 最佳解法思路
- BFS：把“站点”当作 BFS 节点。对每个站点能上哪些线路做映射 `stop -> routes`。
- 从 `source` 出发，每扩一层表示多坐一辆车：遍历当前站点的所有线路，把线路上的所有站点加入下一层。
- 用 `visitedRoute` 避免重复扩展同一条线路。

## Java 最佳实现

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

class Solution {
    public int numBusesToDestination(int[][] routes, int source, int target) {
        if (source == target) return 0;

        Map<Integer, List<Integer>> stopToRoutes = new HashMap<>();
        for (int i = 0; i < routes.length; i++) {
            for (int stop : routes[i]) {
                stopToRoutes.computeIfAbsent(stop, k -> new ArrayList<>()).add(i);
            }
        }

        Deque<Integer> q = new ArrayDeque<>();
        Set<Integer> visitedStop = new HashSet<>();
        boolean[] usedRoute = new boolean[routes.length];

        q.add(source);
        visitedStop.add(source);

        int buses = 0;
        while (!q.isEmpty()) {
            buses++;
            int size = q.size();
            for (int i = 0; i < size; i++) {
                int stop = q.poll();
                List<Integer> rs = stopToRoutes.get(stop);
                if (rs == null) continue;
                for (int rid : rs) {
                    if (usedRoute[rid]) continue;
                    usedRoute[rid] = true;
                    for (int next : routes[rid]) {
                        if (next == target) return buses;
                        if (visitedStop.add(next)) q.add(next);
                    }
                }
            }
        }
        return -1;
    }
}
```
