## 最佳解法思路
- 并查集：
  - 两个字符串相似：要么完全相同，要么恰好有 2 个位置字符不同且可通过交换变相同
  - 枚举任意两字符串，若相似则 union
  - 最后统计并查集根的数量

## Java 最佳实现

```java
import java.util.HashSet;
import java.util.Set;

class Solution {
    static class UF {
        int[] p;
        int[] r;

        UF(int n) {
            p = new int[n];
            r = new int[n];
            for (int i = 0; i < n; i++) p[i] = i;
        }

        int find(int x) {
            if (p[x] != x) p[x] = find(p[x]);
            return p[x];
        }

        void union(int a, int b) {
            int pa = find(a), pb = find(b);
            if (pa == pb) return;
            if (r[pa] < r[pb]) p[pa] = pb;
            else if (r[pa] > r[pb]) p[pb] = pa;
            else {
                p[pb] = pa;
                r[pa]++;
            }
        }
    }

    public int numSimilarGroups(String[] strs) {
        int n = strs.length;
        UF uf = new UF(n);
        for (int i = 0; i < n; i++) {
            for (int j = i + 1; j < n; j++) {
                if (isSimilar(strs[i], strs[j])) uf.union(i, j);
            }
        }
        Set<Integer> roots = new HashSet<>();
        for (int i = 0; i < n; i++) roots.add(uf.find(i));
        return roots.size();
    }

    private boolean isSimilar(String a, String b) {
        int diff = 0;
        int p1 = -1, p2 = -1;
        for (int i = 0; i < a.length(); i++) {
            if (a.charAt(i) == b.charAt(i)) continue;
            diff++;
            if (diff > 2) return false;
            if (p1 == -1) p1 = i;
            else p2 = i;
        }
        if (diff == 0) return true;
        if (diff != 2) return false;
        return a.charAt(p1) == b.charAt(p2) && a.charAt(p2) == b.charAt(p1);
    }
}
```

- 时间：\(O(n^2 \cdot L)\)
- 空间：\(O(n)\)
