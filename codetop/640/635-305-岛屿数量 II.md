## 最佳解法思路
- 并查集（推荐）
- 初始全是水。每次把一个位置变成陆地：岛屿数 +1。
- 与四邻接的已存在陆地做 union，每成功合并一次岛屿数 -1。

## Java 最佳实现

```java
import java.util.ArrayList;
import java.util.List;

class Solution {
    public List<Integer> numIslands2(int m, int n, int[][] positions) {
        UnionFind uf = new UnionFind(m * n);
        boolean[] land = new boolean[m * n];
        int count = 0;
        int[][] dirs = new int[][]{{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

        List<Integer> ans = new ArrayList<>();
        for (int[] p : positions) {
            int r = p[0], c = p[1];
            int idx = r * n + c;
            if (land[idx]) {
                ans.add(count);
                continue;
            }
            land[idx] = true;
            uf.setParent(idx);
            count++;

            for (int[] d : dirs) {
                int nr = r + d[0], nc = c + d[1];
                if (nr < 0 || nr >= m || nc < 0 || nc >= n) continue;
                int nidx = nr * n + nc;
                if (!land[nidx]) continue;
                if (uf.union(idx, nidx)) count--;
            }
            ans.add(count);
        }
        return ans;
    }

    static class UnionFind {
        int[] parent;
        int[] rank;

        UnionFind(int n) {
            parent = new int[n];
            rank = new int[n];
            for (int i = 0; i < n; i++) parent[i] = -1;
        }

        void setParent(int x) {
            parent[x] = x;
            rank[x] = 1;
        }

        int find(int x) {
            while (x != parent[x]) {
                parent[x] = parent[parent[x]];
                x = parent[x];
            }
            return x;
        }

        boolean union(int a, int b) {
            if (parent[a] == -1 || parent[b] == -1) return false;
            int pa = find(a), pb = find(b);
            if (pa == pb) return false;
            if (rank[pa] < rank[pb]) {
                parent[pa] = pb;
            } else if (rank[pa] > rank[pb]) {
                parent[pb] = pa;
            } else {
                parent[pb] = pa;
                rank[pa]++;
            }
            return true;
        }
    }
}
```

- 空间：\(O(mn)\)
