## 最佳解法思路
- 反向处理 + 并查集：先把所有 hits 位置的砖都“先打掉”，在剩余砖块上建并查集，并加一个虚拟顶点 `top` 连接所有顶行砖。
- 然后从后往前把每次打掉的砖“加回来”：加回前后 `top` 连通块大小的变化，就是本次会“重新粘住”的砖数（再减去自己）。

## Java 最佳实现

```java
class Solution {
    public int[] hitBricks(int[][] grid, int[][] hits) {
        int m = grid.length;
        int n = grid[0].length;
        int[][] g = new int[m][n];
        for (int i = 0; i < m; i++) {
            System.arraycopy(grid[i], 0, g[i], 0, n);
        }

        for (int[] h : hits) {
            int r = h[0], c = h[1];
            if (g[r][c] == 1) g[r][c] = 0;
        }

        int top = m * n;
        DSU dsu = new DSU(m * n + 1);
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (g[i][j] != 1) continue;
                int id = i * n + j;
                if (i == 0) dsu.union(id, top);
                if (i > 0 && g[i - 1][j] == 1) dsu.union(id, (i - 1) * n + j);
                if (j > 0 && g[i][j - 1] == 1) dsu.union(id, i * n + (j - 1));
            }
        }

        int[] res = new int[hits.length];
        int[] dr = {1, -1, 0, 0};
        int[] dc = {0, 0, 1, -1};

        for (int i = hits.length - 1; i >= 0; i--) {
            int r = hits[i][0];
            int c = hits[i][1];
            if (grid[r][c] == 0) {
                res[i] = 0;
                continue;
            }

            int before = dsu.size(top);
            int id = r * n + c;
            g[r][c] = 1;

            if (r == 0) dsu.union(id, top);
            for (int k = 0; k < 4; k++) {
                int nr = r + dr[k], nc = c + dc[k];
                if (nr < 0 || nr >= m || nc < 0 || nc >= n) continue;
                if (g[nr][nc] == 1) dsu.union(id, nr * n + nc);
            }

            int after = dsu.size(top);
            res[i] = Math.max(0, after - before - 1);
        }
        return res;
    }

    static class DSU {
        int[] p;
        int[] sz;
        DSU(int n) {
            p = new int[n];
            sz = new int[n];
            for (int i = 0; i < n; i++) {
                p[i] = i;
                sz[i] = 1;
            }
        }
        int find(int x) {
            while (p[x] != x) {
                p[x] = p[p[x]];
                x = p[x];
            }
            return x;
        }
        void union(int a, int b) {
            int ra = find(a), rb = find(b);
            if (ra == rb) return;
            if (sz[ra] < sz[rb]) {
                int t = ra; ra = rb; rb = t;
            }
            p[rb] = ra;
            sz[ra] += sz[rb];
        }
        int size(int x) {
            return sz[find(x)];
        }
    }
}
```

## 复杂度分析
- 时间：\(O((mn + q)\alpha(mn))\)
- 空间：\(O(mn)\)
