## 最佳解法思路
- 若线缆数量 `< n-1`，无论如何都无法连通，返回 `-1`。
- 否则用并查集统计连通分量个数 `cc`，最少操作数为 `cc - 1`。

## Java 最佳实现

```java
class Solution {
    public int makeConnected(int n, int[][] connections) {
        if (connections.length < n - 1) return -1;
        DSU dsu = new DSU(n);
        for (int[] e : connections) {
            dsu.union(e[0], e[1]);
        }
        return dsu.components() - 1;
    }

    static class DSU {
        int[] p;
        int[] r;
        int cc;
        DSU(int n) {
            p = new int[n];
            r = new int[n];
            cc = n;
            for (int i = 0; i < n; i++) p[i] = i;
        }
        int find(int x) {
            while (p[x] != x) {
                p[x] = p[p[x]];
                x = p[x];
            }
            return x;
        }
        void union(int a, int b) {
            int ra = find(a), rb = find(b);
            if (ra == rb) return;
            cc--;
            if (r[ra] < r[rb]) p[ra] = rb;
            else if (r[ra] > r[rb]) p[rb] = ra;
            else {
                p[rb] = ra;
                r[ra]++;
            }
        }
        int components() {
            return cc;
        }
    }
}
```

## 复杂度分析
- 时间：\(O(n + m \alpha(n))\)
- 空间：\(O(n)\)
