## 最佳解法思路
- 思路（推荐）：扫描线 + 线段树（离散化 y）  
对每个矩形产生两条竖边事件：`x1` 加区间 `[y1,y2)`，`x2` 减区间 `[y1,y2)`。  
按 `x` 排序扫描，线段树维护当前所有激活矩形在 y 轴上的“覆盖总长度”。  
相邻两次 x 之间的贡献面积 = `coveredY * (xNext - xCur)`。

## Java 最佳实现

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.List;

class Solution {
    private static final int MOD = 1_000_000_007;

    public int rectangleArea(int[][] rectangles) {
        List<int[]> events = new ArrayList<>();
        int n = rectangles.length;
        int[] ys = new int[n * 2];
        int yi = 0;

        for (int[] r : rectangles) {
            int x1 = r[0], y1 = r[1], x2 = r[2], y2 = r[3];
            events.add(new int[]{x1, y1, y2, 1});
            events.add(new int[]{x2, y1, y2, -1});
            ys[yi++] = y1;
            ys[yi++] = y2;
        }

        Arrays.sort(ys);
        int m = 0;
        for (int v : ys) {
            if (m == 0 || v != ys[m - 1]) {
                ys[m++] = v;
            }
        }
        ys = Arrays.copyOf(ys, m);

        events.sort(Comparator.comparingInt(a -> a[0]));
        SegTree st = new SegTree(ys);

        long area = 0;
        int i = 0;
        while (i < events.size()) {
            int x = events.get(i)[0];
            if (i > 0) {
                int prevX = events.get(i - 1)[0];
                long dx = (long) x - prevX;
                area = (area + st.covered() * dx) % MOD;
            }

            while (i < events.size() && events.get(i)[0] == x) {
                int[] e = events.get(i);
                st.update(e[1], e[2], e[3]);
                i++;
            }
        }

        return (int) (area % MOD);
    }

    private static class SegTree {
        private final int[] ys;
        private final int n; // ys.length-1 segments in y
        private final int[] cover;
        private final long[] len;

        SegTree(int[] ys) {
            this.ys = ys;
            this.n = ys.length - 1;
            this.cover = new int[n * 4 + 5];
            this.len = new long[n * 4 + 5];
        }

        long covered() {
            return n <= 0 ? 0 : len[1];
        }

        void update(int y1, int y2, int delta) {
            if (n <= 0) {
                return;
            }
            int l = Arrays.binarySearch(ys, y1);
            int r = Arrays.binarySearch(ys, y2);
            update(1, 0, n, l, r, delta);
        }

        private void update(int idx, int l, int r, int ql, int qr, int delta) {
            if (ql >= r || qr <= l) {
                return;
            }
            if (ql <= l && r <= qr) {
                cover[idx] += delta;
                pushUp(idx, l, r);
                return;
            }
            int mid = (l + r) >>> 1;
            update(idx << 1, l, mid, ql, qr, delta);
            update(idx << 1 | 1, mid, r, ql, qr, delta);
            pushUp(idx, l, r);
        }

        private void pushUp(int idx, int l, int r) {
            if (cover[idx] > 0) {
                len[idx] = (long) ys[r] - ys[l];
                return;
            }
            if (l + 1 >= r) {
                len[idx] = 0;
                return;
            }
            len[idx] = len[idx << 1] + len[idx << 1 | 1];
        }
    }
}
```

## 复杂度分析
- 时间：\(O(n \log n)\)
- 空间：\(O(n)\)
