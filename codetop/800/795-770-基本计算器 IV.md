## 最佳解法思路
- 思路（推荐）：多项式化简 + 递归下降解析  
把表达式看成多项式：每项是 `系数 * 变量列表`。先用 `evalvars/evalints` 做代入：能代入的变量变成常数。  
用递归下降解析处理 `+ - * ( )`，多项式支持加/减/乘。  
最后把所有项按“次数（变量个数）降序、字典序升序”输出。

## Java 最佳实现

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

class Solution {
    public List<String> basicCalculatorIV(String expression, String[] evalvars, int[] evalints) {
        Map<String, Integer> eval = new HashMap<>();
        for (int i = 0; i < evalvars.length; i++) {
            eval.put(evalvars[i], evalints[i]);
        }

        Parser p = new Parser(expression, eval);
        Poly res = p.parseExpr();
        return res.toList();
    }

    private static class Parser {
        private final String s;
        private final Map<String, Integer> eval;
        private int i = 0;

        Parser(String s, Map<String, Integer> eval) {
            this.s = s;
            this.eval = eval;
        }

        Poly parseExpr() {
            Poly left = parseTerm();
            skip();
            while (i < s.length()) {
                char op = s.charAt(i);
                if (op != '+' && op != '-') {
                    break;
                }
                i++;
                Poly right = parseTerm();
                if (op == '+') {
                    left = left.add(right);
                } else {
                    left = left.sub(right);
                }
                skip();
            }
            return left;
        }

        Poly parseTerm() {
            Poly left = parseFactor();
            skip();
            while (i < s.length() && s.charAt(i) == '*') {
                i++;
                Poly right = parseFactor();
                left = left.mul(right);
                skip();
            }
            return left;
        }

        Poly parseFactor() {
            skip();
            char c = s.charAt(i);
            if (c == '(') {
                i++;
                Poly inside = parseExpr();
                skip();
                i++;
                return inside;
            }

            if (Character.isDigit(c)) {
                int num = 0;
                while (i < s.length() && Character.isDigit(s.charAt(i))) {
                    num = num * 10 + (s.charAt(i) - '0');
                    i++;
                }
                return Poly.constant(num);
            }

            StringBuilder sb = new StringBuilder();
            while (i < s.length() && Character.isLetter(s.charAt(i))) {
                sb.append(s.charAt(i));
                i++;
            }
            String var = sb.toString();
            Integer v = eval.get(var);
            if (v != null) {
                return Poly.constant(v);
            }
            return Poly.variable(var);
        }

        void skip() {
            while (i < s.length() && s.charAt(i) == ' ') {
                i++;
            }
        }
    }

    private static class Poly {
        private final Map<String, Integer> map;

        Poly() {
            this.map = new HashMap<>();
        }

        static Poly constant(int v) {
            Poly p = new Poly();
            if (v != 0) {
                p.map.put("", v);
            }
            return p;
        }

        static Poly variable(String v) {
            Poly p = new Poly();
            p.map.put(v, 1);
            return p;
        }

        Poly add(Poly b) {
            Poly res = new Poly();
            res.map.putAll(this.map);
            for (Map.Entry<String, Integer> e : b.map.entrySet()) {
                res.map.put(e.getKey(), res.map.getOrDefault(e.getKey(), 0) + e.getValue());
            }
            res.clean();
            return res;
        }

        Poly sub(Poly b) {
            Poly res = new Poly();
            res.map.putAll(this.map);
            for (Map.Entry<String, Integer> e : b.map.entrySet()) {
                res.map.put(e.getKey(), res.map.getOrDefault(e.getKey(), 0) - e.getValue());
            }
            res.clean();
            return res;
        }

        Poly mul(Poly b) {
            Poly res = new Poly();
            for (Map.Entry<String, Integer> ea : this.map.entrySet()) {
                for (Map.Entry<String, Integer> eb : b.map.entrySet()) {
                    String term = mergeTerm(ea.getKey(), eb.getKey());
                    int coef = ea.getValue() * eb.getValue();
                    res.map.put(term, res.map.getOrDefault(term, 0) + coef);
                }
            }
            res.clean();
            return res;
        }

        private void clean() {
            map.entrySet().removeIf(e -> e.getValue() == 0);
        }

        List<String> toList() {
            List<String> terms = new ArrayList<>(map.keySet());
            Collections.sort(terms, (a, b) -> {
                int da = degree(a);
                int db = degree(b);
                if (da != db) {
                    return db - da;
                }
                return a.compareTo(b);
            });

            List<String> res = new ArrayList<>();
            for (String t : terms) {
                int coef = map.get(t);
                if (t.length() == 0) {
                    res.add(String.valueOf(coef));
                } else {
                    res.add(coef + "*" + t);
                }
            }
            return res;
        }

        private int degree(String term) {
            if (term.length() == 0) {
                return 0;
            }
            int d = 1;
            for (int i = 0; i < term.length(); i++) {
                if (term.charAt(i) == '*') {
                    d++;
                }
            }
            return d;
        }

        private String mergeTerm(String a, String b) {
            if (a.length() == 0) {
                return b;
            }
            if (b.length() == 0) {
                return a;
            }
            String[] aa = a.split("\\*");
            String[] bb = b.split("\\*");
            String[] all = new String[aa.length + bb.length];
            System.arraycopy(aa, 0, all, 0, aa.length);
            System.arraycopy(bb, 0, all, aa.length, bb.length);
            Arrays.sort(all);
            return String.join("*", all);
        }
    }
}
```
